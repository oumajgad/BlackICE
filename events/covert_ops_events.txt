country_event = {
	id = 1550
	is_triggered_only = yes
	title = "Let's build a level 2 supply port for our invasion force!"
	desc = "EVTDESC1550"
	picture = overlord-allies-reinforcement-location
	option = {
		name = "Our troops need supply!"
		any_controlled = {
			Limit = {
				Has_province_flag = invasion_port_province_marked
			}
			clr_province_flag = invasion_port_province_marked
			set_province_flag = invasion_port_province_built
			set_province_flag = invasion_port_lvl_2
			naval_base = 2
			add_province_modifier = {
				name = invasion_port_cooldown
				duration = 180
			}
			add_province_modifier = {
				name = invasion_port_province_30_days
				duration = 20
			}
		}
		supplies = -15000
		clr_country_flag = invasion_port_marked
		set_country_flag = invasion_port_built
		Add_country_modifier = {
			Name = invasion_port_timer
			Duration = 35
		}
	}
	option = {
		name = "I've changed my mind"
	}
}

# The player's own resource conversion, fired by the covert operation
# set_resource_conversion in the capital. The variables are read by BiceLib's
# customResourceConversion (DaveStuff/luabinaries/bice/BiceLib/BiceLib/Hooks/ResourceConversion.hpp):
# a day's amount, capped at what the country's IC allows. The utility's Resource
# Conversion page sets the same variables to any value, and the limits beside them.
# Every value is in stockpile units, and 0 means off: a save drops a variable that is 0.

# Which conversion to set

	id = 1553
	is_triggered_only = yes
	title = "EVTNAME1553"
	desc = "EVTDESC1553"
	picture = "industry"
	option = {
		name = "EVTOPTA1553"
		country_event = 1554
	}
	option = {
		name = "EVTOPTB1553"
		country_event = 1555
	}
	option = {
		name = "EVTOPTC1553"
	}
}
